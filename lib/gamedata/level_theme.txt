# file: level_theme.txt

# Named themes for specific dungeon levels.  These can provide a sidebar label
# and, as the theme system expands, generation rules for distinctive floors.

# theme: named generation theme
# spawn: exact depth : Classic chance : Recall chance : Nightmare chance
# label: sidebar label to show below the depth; keep to 13 characters or fewer
# label-color: Angband color character
# seed-room-tag:room-template tag:percent chance
# seed-terrain:terrain feature code
# unique:unique monster to place on the level if still alive
# unique-block-down-stairs:unique monster to place on the level if still alive;
#   while present, generated down stairs are replaced with up stairs, and a down
#   stair appears when the unique is slain
